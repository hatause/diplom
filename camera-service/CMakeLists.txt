cmake_minimum_required(VERSION 3.20)
project(aituto_camera_service VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CAMERA_WITH_OPENCV "Enable real camera snapshots (requires OpenCV)" ON)
include(FetchContent)
set(HTTPLIB_REQUIRE_OPENSSL OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(httplib URL https://codeload.github.com/yhirose/cpp-httplib/tar.gz/refs/tags/v0.20.0 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(httplib json)
add_executable(camera-service src/main.cpp src/CameraManager.cpp src/ApiServer.cpp)
target_link_libraries(camera-service PRIVATE httplib::httplib nlohmann_json::nlohmann_json)
target_compile_options(camera-service PRIVATE -Wall -Wextra -Wpedantic)
if(CAMERA_WITH_OPENCV)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgcodecs videoio)
  target_compile_definitions(camera-service PRIVATE CAMERA_WITH_OPENCV=1)
  target_include_directories(camera-service PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(camera-service PRIVATE ${OpenCV_LIBS})
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME camera_http_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/contract.py $<TARGET_FILE:camera-service>)
endif()
